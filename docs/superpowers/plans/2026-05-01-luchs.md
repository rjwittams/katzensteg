# Luchs Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `luchs`, an SDL-backed local HTML viewer that renders through system Chromium/CDP and presents frames through SDL for Katzensteg capture.

**Architecture:** Keep the first slice in `tools/luchs/` and make it independent of Katzensteg internals. A small CDP helper launches or connects to a Chromium-family browser, captures frames, and sends framed PNG/RGBA data to a Zig SDL presenter that updates one SDL texture and calls `SDL_RenderPresent`.

**Tech Stack:** Zig 0.15.2, SDL2, Node.js 18+, Chrome DevTools Protocol, system Chromium/Chrome/Brave/Edge, optional tiny PNG decoder such as `stb_image`.

```zig
const std = @import("std");

const CliOptions = struct {
    html_path: []const u8,
};

fn parseArgs(args: []const []const u8) !CliOptions {
    if (args.len != 2) return error.Usage;
    return .{ .html_path = args[1] };
}

test "luchs requires one html path" {
    try std.testing.expectError(error.Usage, parseArgs(&.{"luchs"}));
    const opts = try parseArgs(&.{ "luchs", "tools/luchs/testdata/static.html" });
    try std.testing.expectEqualStrings("tools/luchs/testdata/static.html", opts.html_path);
}

pub fn main() !void {
    var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
    defer arena.deinit();
    const args = try std.process.argsAlloc(arena.allocator());
    _ = try parseArgs(args);
}
```

```zig
fn fillTestFrame(buf: []u8, width: usize, height: usize, tick: usize) void {
    var y: usize = 0;
    while (y < height) : (y += 1) {
        var x: usize = 0;
        while (x < width) : (x += 1) {
            const i = (y * width + x) * 4;
            buf[i + 0] = @intCast((x + tick) % 256);
            buf[i + 1] = @intCast((y + tick) % 256);
            buf[i + 2] = 160;
            buf[i + 3] = 255;
        }
    }
}

test "fillTestFrame writes rgba pixels" {
    var buf: [8]u8 = undefined;
    fillTestFrame(&buf, 1, 2, 3);
    try std.testing.expectEqual(@as(u8, 3), buf[0]);
    try std.testing.expectEqual(@as(u8, 3), buf[1]);
    try std.testing.expectEqual(@as(u8, 255), buf[3]);
}
```
